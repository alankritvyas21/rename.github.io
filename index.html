<script>

async function renameAndZip() {

  const files = document.getElementById("files").files;

  const prefix = document.getElementById("prefix").value.trim();

  let number = parseInt(document.getElementById("startNumber").value);

  if (!files.length || !prefix || isNaN(number)) {

    alert("Please select files and fill all fields.");

    return;

  }

  const zip = new JSZip();

  for (let i = 0; i < files.length; i++) {

    const file = files[i];

    const ext = file.name.substring(file.name.lastIndexOf("."));

    const newName = `${prefix} (${number})${ext}`;

    zip.file(newName, file);

    number++;

  }

  const content = await zip.generateAsync({ type: "blob" });

  saveAs(content, "renamed-images.zip");

}
